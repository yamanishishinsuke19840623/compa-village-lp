━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━
  COMPA VILLAGE 予約管理システム デプロイ手順書
  （やまちゃん作業用）
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

目的：
  LPの予約フォームから送信されたリクエストを
  ①スプレッドシート台帳に記録　②リアルタイム空室判定
  ③jannenenenene@gmail.com（ゆうさん）へ通知メール
  が自動で動くようにする。

作業時間：約10分
必要なもの：やまちゃんの Google アカウント
  （claspでの直接デプロイはネットワーク制限で失敗したため、
   この手順書での手動デプロイに切り替えています）

━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━
  STEP 1　Google Apps Script を開く
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

ブラウザで以下を開く:
  https://script.google.com

「新しいプロジェクト」をクリック


━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━
  STEP 2　コードを貼り付ける
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

1. 左側のファイル欄に「コード.gs」が表示されている
2. その中身を全部削除する
3. 「Code.gs」（このフォルダの中）の中身を全コピーして貼り付ける
4. Ctrl+S で保存（プロジェクト名は「COMPA VILLAGE 予約管理」などにする）


━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━
  STEP 3　台帳スプレッドシートを作る（初回のみ）
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

1. 上部の関数選択プルダウンで「setupSheet」を選ぶ
2. 「実行」ボタン（▷）をクリック
3. 初回は権限確認が出るので「権限を確認」→ Googleアカウントを選択
   →「詳細」→「（安全ではないページ）に移動」→「許可」
4. 下部の「実行ログ」に以下が表示される：

     Sheet ID: 1AbCdEfGhIjKlMnOpQrStUvWxYz...
     Sheet URL: https://docs.google.com/spreadsheets/d/...

   ★ この Sheet ID をコピーしておく ★

5. コード冒頭の以下の行に貼り付けて、再度 Ctrl+S で保存：

     var SHEET_ID = ''; ← ここにSTEP3でコピーしたIDを貼る


━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━
  STEP 4　ウェブアプリとしてデプロイする
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

1. 右上の「デプロイ」ボタン → 「新しいデプロイ」をクリック
2. 「種類の選択」→「ウェブアプリ」を選択
3. 以下のように設定する:

   ┌─────────────────────────────────┐
   │ 次のユーザーとして実行：自分（自分のGmailアドレス） │
   │ アクセスできるユーザー：全員                      │
   └─────────────────────────────────┘

4. 「デプロイ」ボタンをクリック
5. 権限の確認画面が出たら STEP3と同様に許可する

6. デプロイ完了！「ウェブアプリの URL」が表示される
   例：
     https://script.google.com/macros/s/AKfy.../exec

   ★ この URL をコピーしておく ★


━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━
  STEP 5　LP の HTML に URL を設定する
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

index.html の以下の行を探して URL を貼り付ける:

  var GAS_BOOKING_ENDPOINT = 'YOUR_GAS_ENDPOINT_HERE';
                              ↑ ここをSTEP4でコピーしたURLに差し替え

その後、git add → commit → push してデプロイ完了。


━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━
  STEP 6　動作確認
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

公開LPの「直接予約」セクションで日程を選び、テスト予約リクエストを送信して:

  ① 予約者のメールに「リクエストを受け付けました」メールが届く
  ② jannenenenene@gmail.com に通知が届く
  ③ やまちゃんの Gmail にも CC で届く
  ④ 台帳スプレッドシートに行が追加される
  ⑤ 同じ日程・同じ部屋タイプで再度予約しようとすると
     「満室」扱いになり、予約リクエストできない

以上で完了！


━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━
  予約の確定について
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

スプレッドシートを開くと、上部メニューに
「🌵 COMPA予約」というカスタムメニューが追加されています。

  1. 確定させたい予約の行をクリック（どのセルでもOK）
  2. 「🌵 COMPA予約」→「選択行を『確定』にして確定メールを送信」
  3. お客様に確定メールが自動送信され、
     ステータス列が「確定」に変わります


━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━
  注意点
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

・部屋数は「各タイプ1部屋」という前提でコードを書いています
  （コード冒頭の ROOM_CAPACITY）。
  実際の部屋数が違う場合は、この数字を修正してください。

・メールはやまちゃんの Google アカウントから送信されます
  （送信者欄がやまちゃんのアドレスになる場合がある）
  → 返信先（Reply-To）は jannenenenene@gmail.com に設定済み

・GAS の無料枠：メール送信 100件/日。問題になることはまずなし

・GAS のコードを変更したら「デプロイ管理」から
  「新バージョンとして再デプロイ」すること

・現時点では決済（Stripe）は未接続です。予約はあくまで
  「リクエスト」で、ゆうさんが手動で確定させる流れになります。
  Stripeの支払いリンクが揃ったら（別紙 Stripe手順書 参照）、
  確定メールに決済リンクを添える形に拡張できます。
